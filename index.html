import os from datetime import datetime from flask import Flask, request, jsonify, render_template from flask_sqlalchemy import SQLAlchemy # Fondamentale per il database app = Flask(__name__) # --- CONFIGURAZIONE PERCORSI E CARTELLE --- # Cartella per i file di testo Pegaso EXPORT_FOLDER = 'esportazioni_pegaso' if not os.path.exists(EXPORT_FOLDER): os.makedirs(EXPORT_FOLDER) # Percorso per il database SQLite basedir = os.path.abspath(os.path.dirname(__file__)) app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///' + os.path.join(basedir, 'gestionale.db') app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False # Inizializziamo il database db = SQLAlchemy(app) # --- MODELLO DATABASE --- # Questa classe definisce come verranno salvati i dati nel file gestionale.db class Ospite(db.Model): id = db.Column(db.Integer, primary_key=True) # Identificativo unico progressivo nome = db.Column(db.String(80)) cognome = db.Column(db.String(80)) documento = db.Column(db.String(20)) data_nascita = db.Column(db.String(20)) # Nuovo: salva la data letta dall'OCR sesso = db.Column(db.String(1)) # Nuovo: M o F cittadinanza = db.Column(db.String(50)) # Nuovo: es. ITALIA data_arrivo = db.Column(db.DateTime, default=datetime.utcnow) # Data di sistema # Crea il database se non esiste with app.app_context(): db.create_all() # --- ROTTE --- @app.route('/') def home(): # Carica la tua pagina principale index.html return render_template('index.html') @app.route('/salva_checkin', methods=['POST']) def salva_checkin(): try: # Recuperiamo i dati inviati dal modulo (nome, cognome, ecc.) dati = request.json # 1.
